cmake_minimum_required(VERSION 3.15)
project(CppMultiLLMChat VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(BUILD_CHAT_SERVER "Build the HTTP chat server" ON)
option(BUILD_INTERACTIVE_TESTS "Build manual tests that require live model services" OFF)

include(GNUInstallDirs)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
pkg_check_modules(FMT REQUIRED IMPORTED_TARGET fmt)
pkg_check_modules(SPDLOG REQUIRED IMPORTED_TARGET spdlog)
pkg_check_modules(SQLITE3 REQUIRED IMPORTED_TARGET sqlite3)

find_path(CPPHTTPLIB_INCLUDE_DIR NAMES httplib.h)
if(NOT CPPHTTPLIB_INCLUDE_DIR)
    message(FATAL_ERROR "cpp-httplib header httplib.h was not found; provide CPPHTTPLIB_INCLUDE_DIR.")
endif()
add_library(cpp_httplib_headers INTERFACE)
target_include_directories(cpp_httplib_headers INTERFACE "${CPPHTTPLIB_INCLUDE_DIR}")
target_compile_definitions(cpp_httplib_headers INTERFACE CPPHTTPLIB_OPENSSL_SUPPORT)
target_link_libraries(cpp_httplib_headers INTERFACE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
    target_link_libraries(cpp_httplib_headers INTERFACE ws2_32 crypt32)
endif()

add_subdirectory(sdk)
if(BUILD_CHAT_SERVER)
    add_subdirectory(apps/chat_server)
endif()
if(BUILD_INTERACTIVE_TESTS)
    add_subdirectory(tests)
endif()
